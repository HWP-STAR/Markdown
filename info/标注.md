# labelme
pip install labelme
labelme

# 转换
pip install yolo-to-labelme
yolotolabelme

# 使用
yolotolabelme \
  --yolo labels \
  --labelme labelme_json \
  --classes classes.txt \
  --width 2560 --height 1440

# 打开
labelme dataset/labelme_json --nodata
labelme labelme_json --nodata
--nodata 是 LabelMe 的启动参数，意思是：
保存 JSON 标注文件时，不要把原图以 base64 形式塞进 JSON 里。


pip install labelme2yolo
labelme2yolo --json_dir dataset/labelme_json
