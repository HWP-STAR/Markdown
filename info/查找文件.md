# 文件查找
1. find（全盘 / 目录精准搜索，最常用）
bash
运行
# 在根目录全盘搜索文件名包含xxx
find / -name "xxx" 2>/dev/null

# 在当前目录递归查找，大小写不敏感
find . -iname "filename"

# 只查找普通文件，排除文件夹
find /home -type f -name "*.log"
2>/dev/null 屏蔽权限不足的报错。

3. grep 查找文件内容（按文本内容搜文件）
bash
运行
# 在当前目录所有文件里搜索包含"关键词"的文件
grep -r "关键词" .

# 只列出文件名，不显示匹配行
grep -rl "关键词" .
